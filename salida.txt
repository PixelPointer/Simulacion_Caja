Efectos de utilizar diferentes semillas aleatorias:

Al utilizar una semilla diferente se obtienen nuevas secuencias y, consecuentemente, pueden
generarse distintos tiempos entre llegadas y tiempos de servicio.

Los resultados que permanecen aproximadamente estables al utilizar diferentes semillas
aleatorias son las medias de los tiempos entre llegadas y de los tiempos de servicio. Esto se puede
comprobar mediante la comparación de dos ejecuciones con semillas distintas. En la primera
ejecución, utilizando la semilla 101, se obtuvo una media de tiempo entre llegadas de 2.8 y una
media de tiempo de servicio de 2.4. Por otro lado, en la segunda ejecución, con la semilla 573, se
obtuvo una media de tiempo entre llegadas de 2.5 y una media de tiempo de servicio de 2.1.

Al comparar ambos resultados, se observa que los valores de las medias son relativamente
cercanos, sin presentar diferencias significativas. Esto indica que, aunque la semilla aleatoria
modifica los valores individuales generados en cada simulación, las medias tienden a mantenerse
aproximadamente estables.

 1. Diseño de la estructura del sistema de atención
 El sistema es una cola de un solo servidor con disciplina FIFO. Cada cliente recorre el
 mismo camino: llega, espera si la caja está ocupada, es atendido y sale.




             Elemento                                            Diseño
  Entidad                  Cliente
  Recursos                 1 servidor (caja) y una cola de capacidad ilimitada
  Disciplina               FIFO: el primero en llegar es el primero en ser atendido
  Eventos                  LLEGADA y FIN de servicio (simulación por eventos discretos)
  Estado inicial           Caja libre, cola vacía, reloj en 0
  Unidad de tiempo         Minutos
  Salida de cada réplica   Espera media y espera máxima de los 100 clientes



 2. Configuración de la entidad Cliente
 El cliente se representa con una clase que guarda los tiempos que se le asignan al generarlo
 (tiempo entre llegadas y tiempo de servicio) y las horas que se registran durante la
 simulación (llegada, inicio y fin). La espera no se guarda: se calcula como inicio − llegada.

            class Cliente:
                   def __init__(self, ident, t_entre_llegadas, t_servicio):

self.t_entre = t_entre_llegadas                      # tiempo desde el cliente anterior

                   self.fin = None


              @property
              def espera(self):
                   return self.inicio - self.llegada

def acumulada(dic):
   """Devuelve (valores, F(x)) de la distribucion empirica."""
    total = sum(dic.values())
    xs = sorted(dic)
    F, acum = [], 0.0
    for x in xs:
         acum += dic[x] / total
         F.append(acum)
    return xs, F



XS_LLEG, F_LLEG = acumulada(LLEGADAS)
XS_SERV, F_SERV = acumulada(SERVICIO)



def inversa(u, xs, F):
    """Transformada inversa discreta: primer x con u <= F(x)."""
    for x, f in zip(xs, F):
          if u <= f + 1e-12:
                   return x
    return xs[-1]

6. Lógica de llegada, espera, atención y salida
El reloj avanza de evento en evento. Los eventos futuros se guardan en un heap ordenado por
tiempo, y en cada paso se procesa el más próximo.

                                                                  ocurre

                          está libre, el cliente pasa a atención con espera 0; si no, se une a la cola FIFO.

 FIN de servicio          El cliente sale del sistema (se registra su hora de fin). La caja queda libre y, si
                          hay cola, entra a atención el primer cliente de la fila.

import math, bisect
from scipy.stats import chi2, norm

class CongruencialAditivo:
    def __init__(self, semilla, j=24, k=55, m=2**31):
        self.j, self.k, self.m = j, k, m
        x, self.hist = semilla, []
        for _ in range(k):
            x = (1103515245 * x + 12345) % 2**31
            self.hist.append(x)

    def siguiente(self):
        x = (self.hist[-self.j] + self.hist[-self.k]) % self.m
        self.hist.append(x); self.hist.pop(0)
        return x

    def random(self):
        return self.siguiente() / self.m

llegadas = {1: 0.10, 2: 0.20, 3: 0.30, 4: 0.30, 5: 0.10}
servicio = {1: 0.20, 2: 0.40, 3: 0.30, 4: 0.10}

def construir_cdf(dist):
    valores, acum, s = [], [], 0
    for v, p in dist.items():
        s += p
        valores.append(v); acum.append(round(s, 10))
    return valores, acum

def inversa(u, valores, acum):
    return valores[bisect.bisect_right(acum, u)]

cdf_a, cdf_s = construir_cdf(llegadas), construir_cdf(servicio)

def generar(n, semilla):
    ga = CongruencialAditivo(semilla)
    gs = CongruencialAditivo(semilla + 1000)
    ua = [ga.random() for _ in range(n)]
    us = [gs.random() for _ in range(n)]
    ta = [inversa(u, *cdf_a) for u in ua]
    ts = [inversa(u, *cdf_s) for u in us]
    return ua, ta, us, ts

def prueba_media(u, alfa=0.05):
    n = len(u); xbar = sum(u) / n
    z0 = (xbar - 0.5) * math.sqrt(n) / math.sqrt(1/12)
    zc = norm.ppf(1 - alfa/2)
    li = 0.5 - zc / math.sqrt(12 * n)
    ls = 0.5 + zc / math.sqrt(12 * n)
    return {"media": xbar, "Z0": z0, "Zcrit": zc, "lim_inf": li, "lim_sup": ls,
"acepta": li <= xbar <= ls}
def prueba_varianza(u, alfa=0.05):
    n = len(u); xbar = sum(u) / n
    s2 = sum((x - xbar)**2 for x in u) / (n - 1)
    li = chi2.ppf(alfa/2, n - 1) / (12 * (n - 1))
    ls = chi2.ppf(1 - alfa/2, n - 1) / (12 * (n - 1))
    return {"varianza": s2, "lim_inf": li, "lim_sup": ls, "acepta": li <= s2 <=
ls}

def prueba_corridas(u, alfa=0.05):
    n = len(u)
    s = [1 if u[i+1] > u[i] else 0 for i in range(n - 1)]
    c0 = 1 + sum(1 for i in range(1, len(s)) if s[i] != s[i-1])
    mu = (2*n - 1) / 3
    var = (16*n - 29) / 90
    z0 = (c0 - mu) / math.sqrt(var)
    zc = norm.ppf(1 - alfa/2)
    return {"C0": c0, "mu_C": mu, "var_C": var, "Z0": z0, "Zcrit": zc, "acepta":
abs(z0) < zc}

def decision(p):
    return "No se rechaza" if p["acepta"] else "SE RECHAZA"

def reporte_pruebas(nombre, u, alfa=0.05):
    pm, pv, pc = prueba_media(u, alfa), prueba_varianza(u, alfa),
prueba_corridas(u, alfa)
    print(f"\n___Flujo: {nombre} (n = {len(u)}, alfa = {alfa})___")
    print(f"{'Prueba':<22}{'Estadístico':<28}{'Criterio de
aceptación':<30}{'Decisión'}")
    print("-" * 95)
    print(f"{'Media':<22}{'media = %.4f' % pm['media']:<28}"
          f"{'%.4f <= media <= %.4f' % (pm['lim_inf'],
pm['lim_sup']):<30}{decision(pm)}")
    print(f"{'Varianza':<22}{'s2 = %.4f' % pv['varianza']:<28}"
          f"{'%.4f <= s2 <= %.4f' % (pv['lim_inf'],
pv['lim_sup']):<30}{decision(pv)}")
    print(f"{'Corridas arriba/abajo':<22}{'C_0 = %d, Z_0 = %.3f' % (pc['C0'],
pc['Z0']):<28}"
          f"{'|Z_0| < %.3f' % pc['Zcrit']:<30}{decision(pc)}")
